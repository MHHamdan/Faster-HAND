% ---------------------------------------------------------------------------
% HAND architecture figure -- reproducible TikZ source for a single IEEE
% column (\columnwidth, ~88.9 mm). Drawn at natural size: no \resizebox, so
% every label is set at the size it is printed (\scriptsize = 7 pt, secondary
% labels 6 pt; nothing smaller than 6 pt).
%
% Reading order: (1) encoder column, (2) decoder column, (3) output stream,
% then the two OPTIONAL mechanisms A and B in the dashed band at the bottom.
% Markers A and B on the main diagram show where each mechanism attaches.
%
% Every block corresponds to code in this repository:
%   conv / DS blocks         hand/models/baseline/fcn_encoder.py   (FCN_Encoder)
%   2D PE + flattening       hand/models/baseline/attention.py     (FeaturesUpdater)
%   decoder layer            hand/models/baseline/attention.py     (GlobalDecoderLayer)
%   decoder stack, shared KV hand/models/baseline/attention.py     (GlobalAttDecoder)
%   embedding, 1D PE, head   hand/models/baseline/dan_decoder.py   (GlobalHTADecoder)
%   draft heads              hand/models/baseline/spec_heads.py    (SpeculativeHeads)
%   draft/verify/rollback    release/hand_release/inference.py     (_decode_speculative)
%
% Shapes are the trained page configuration (release/hand-read2016-page/config.json):
% d = 256, 8 decoder layers, 4 heads, FFN width 256, self-attention band w = 100,
% output |V| = 100. Layout tokens are the circled letters of the charset:
% p/P page, n/N page number, s/S section, b/B body, a/A annotation
% (lower case opens an element, upper case closes it).
%
% Palette: Okabe-Ito (colour-blind safe). Colour is never the only cue: base
% HAND uses solid borders, optional mechanisms dashed borders + letter tags.
%
% Requires \usepackage{tikz} with libraries arrows, positioning, fit, calc,
% backgrounds (declared in the manuscript preamble). No further libraries.
% ---------------------------------------------------------------------------
\providecolor{hndBlue}{RGB}{0,114,178}      % Okabe-Ito blue        : visual encoder
\providecolor{hndPurple}{RGB}{204,121,167}  % Okabe-Ito red-purple  : feature grid / memory
\providecolor{hndNavy}{RGB}{28,46,98}       % dark blue             : base decoder
\providecolor{hndGreen}{RGB}{0,158,115}     % Okabe-Ito green       : joint text+layout output
\providecolor{hndOrange}{RGB}{230,159,0}    % Okabe-Ito orange      : A, shared visual K/V
\providecolor{hndVerm}{RGB}{213,94,0}       % Okabe-Ito vermillion  : B, speculative decoding
%
% Layout-token chips, usable in captions and text as well:
%   \ltb{p} = opening token (circled lower case), \lte{P} = closing token.
\providecommand{\ltb}[1]{\tikz[baseline=(c.base)]\node[draw=hndGreen!80!black,%
  fill=white, circle, inner sep=0.2pt, minimum size=2.3mm, line width=0.4pt,%
  font=\fontsize{6}{6}\selectfont\sffamily, text=hndGreen!60!black] (c) {#1};}%
\providecommand{\lte}[1]{\tikz[baseline=(c.base)]\node[draw=hndGreen!80!black,%
  fill=hndGreen!22, circle, inner sep=0.2pt, minimum size=2.3mm, line width=0.4pt,%
  font=\fontsize{6}{6}\selectfont\sffamily\bfseries, text=hndGreen!60!black] (c) {#1};}%
%
\begin{tikzpicture}[
